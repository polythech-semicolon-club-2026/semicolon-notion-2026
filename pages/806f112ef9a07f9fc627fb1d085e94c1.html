  function setTopic(el, topic) {
    document.querySelectorAll('.chip').forEach(c => c.classList.remove('active'));
    el.classList.add('active');
    currentTopic = topic;
    document.getElementById('input-box').placeholder = `${topic}에 대해 질문하세요...`;
    document.getElementById('input-box').focus();
  }

  function autoResize(el) {
    el.style.height = 'auto';
    el.style.height = Math.min(el.scrollHeight, 100) + 'px';
  }

  function handleKey(e) {
    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMsg(); }
  }

  function removeEmpty() {
    const e = document.getElementById('empty-state');
    if (e) e.remove();
  }

  function addMsg(role, text) {
    removeEmpty();
    const box = document.getElementById('chat-box');
    const row = document.createElement('div');
    row.className = `msg-row ${role === 'user' ? 'user' : ''}`;

    const av = document.createElement('div');
    av.className = `avatar ${role === 'user' ? 'user' : 'ai'}`;
    av.textContent = role === 'user' ? '🧑' : '🦅';

    const bub = document.createElement('div');
    bub.className = `bubble ${role === 'user' ? 'user' : 'ai'}`;

    if (role === 'assistant') {
      bub.innerHTML = formatText(text);
    } else {
      bub.textContent = text;
    }

    row.appendChild(av);
    row.appendChild(bub);
    box.appendChild(row);
    box.scrollTop = box.scrollHeight;
    return bub;
  }

  function formatText(t) {
    const parts = t.split(/(```[\s\S]*?```)/g);
    return parts.map(p => {
      if (p.startsWith('```')) {
        const code = p.replace(/```\w*\n?/,'').replace(/```$/,'').trim();
        return `<pre>${esc(code)}</pre>`;
      }
      return esc(p).replace(/\n/g,'<br>');
    }).join('');
  }

  function esc(s) {
    return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  }

  function showTyping() {
    removeEmpty();
    const box = document.getElementById('chat-box');
    const row = document.createElement('div');
    row.className = 'msg-row';
    row.id = 'typing-row';
    const av = document.createElement('div');
    av.className = 'avatar ai';
    av.textContent = '🦅';
    const bub = document.createElement('div');
    bub.className = 'bubble ai';
    bub.innerHTML = '<div class="typing-dots"><span></span><span></span><span></span></div>';
    row.appendChild(av); row.appendChild(bub);
    box.appendChild(row);
    box.scrollTop = box.scrollHeight;
  }

  function removeTyping() {
    const t = document.getElementById('typing-row');
    if (t) t.remove();
  }

  async function sendMsg() {
    if (isLoading) return;
    const inp = document.getElementById('input-box');
    const text = inp.value.trim();
    if (!text) return;

    inp.value = '';
    inp.style.height = 'auto';
    isLoading = true;
    document.getElementById('send-btn').disabled = true;

    addMsg('user', text);
    history.push({ role: 'user', content: text });
    showTyping();

    try {
      const sysFull = SYSTEM + (currentTopic ? `\n\n현재 질문 주제: ${currentTopic}` : '');
      const geminiMessages = history.map(m => ({
        role: m.role === 'assistant' ? 'model' : 'user',
        parts: [{ text: m.content }]
      }));
      const res = await fetch(
        `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${API_KEY}`,
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            system_instruction: { parts: [{ text: sysFull }] },
            contents: geminiMessages,
            generationConfig: { maxOutputTokens: 1000 }
          })
        }
      );
      const data = await res.json();
      removeTyping();

      if (data.candidates && data.candidates[0]) {
        const reply = data.candidates[0].content.parts[0].text;
        history.push({ role: 'assistant', content: reply });
        addMsg('assistant', reply);
      } else {
        console.error(data);
        addMsg('assistant', '잠시 날개를 접겠습니다. 다시 시도해주세요. (오류: ' + (data.error?.message || '알 수 없음') + ')');
      }
    } catch(e) {
      removeTyping();
      addMsg('assistant', '연결이 끊겼습니다. 철새처럼 잠시 후 다시 돌아올게요. (' + e.message + ')');
    }

    isLoading = false;
    document.getElementById('send-btn').disabled = false;
    inp.focus();
  }
